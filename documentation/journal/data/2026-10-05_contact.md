```cpp
#include <cstdio>
#include <cmath>
#include "mbd/forces/plane_contact.hpp"
#include "mbd/kernel/simulator.hpp"
using namespace mbd; using namespace mbd::kernel;
int main(){
  const Real m=10, k=1e5, c=1000, mu=0.5, vs=1e-3, hx=0.2, hy=0.15, hz=0.1;
  for (Real dth : {-1.0, 1.0, 3.0}) {
    const Real th = std::atan(mu) + dth*pi/180;
    System sys; sys.model.gravity = g_accel*Vec3(-std::sin(th),0,-std::cos(th));
    int b = sys.model.add_body(0,std::make_shared<FreeJointModel>(),Transform3(),Transform3(),RigidBodyInertia::from_solid_box(m,Vec3(hx,hy,hz)));
    PlaneContactParams p; p.stiffness=k; p.damping=c; p.friction=mu; p.slip_speed=vs; p.damping_depth=1e-4;
    std::vector<ContactSphere> pts; for(int i:{-1,1}) for(int j:{-1,1}) pts.push_back({Vec3(i*hx,j*hy,-hz),0.0});
    auto pc=std::make_shared<PlaneContact>(b,pts,p); sys.force_elements.push_back(pc);
    Simulator sim(sys); const Real d0 = m*g_accel*std::cos(th)/(4*k); sim.q(2) = hz - d0; sim.refresh();
    const Real dt=2e-4; Real v05=0;
    for (int n=1;n<=5000;++n){ sim.step(dt); if(n==2500) v05=sim.states()[1].v_WB.x(); }
    const Real v1=sim.states()[1].v_WB.x();
    const Real a_th = -g_accel*(std::sin(th)-mu*std::cos(th));
    if (dth>0) printf("incline %+.0f deg: acceleration %.10f theory %.10f rel err %.3g\n", dth, (v1-v05)/0.5, a_th, std::abs((v1-v05)/0.5-a_th)/std::abs(a_th)); else { Real cr=-vs*std::atanh(std::tan(th)/mu); printf("incline %+.0f deg: creep %.10f theory %.10f rel err %.3g (at 0.5 s %.3g)\n", dth, v1, cr, std::abs(v1-cr)/std::abs(cr), std::abs(v05-cr)/std::abs(cr)); }
  }
}
```

```cpp
#include <cstdio>
#include <cmath>
#include "mbd/forces/plane_contact.hpp"
#include "mbd/forces/force_element.hpp"
#include "mbd/kernel/simulator.hpp"
#include "mbd/kernel/statics.hpp"
using namespace mbd; using namespace mbd::kernel;
Real rel(Real a, Real b){return std::abs(a-b)/std::abs(b);}
int main(){
 { const Real m=2,r=0.1,k=5e6,e=1.5; System sys; int b=sys.model.add_body(0,std::make_shared<FreeJointModel>(),Transform3(),Transform3(),RigidBodyInertia::from_solid_box(m,Vec3(0.07,0.07,0.07))); PlaneContactParams p; p.stiffness=k; p.exponent=e; p.friction=0.8; sys.force_elements.push_back(std::make_shared<PlaneContact>(b,std::vector<ContactSphere>{{Vec3::Zero(),r}},p)); Simulator sim(sys); sim.q(2)=r-1e-4; auto rep=static_equilibrium(sim); Real depth=std::pow(m*g_accel/k,1/e);
   printf("hertz: depth %.6e error %.3g bound %.3g iterations %d neutral %d\n", depth, std::abs((r-sim.q(2))-depth), m*1e-6/(e*k*std::pow(depth,e-1)), rep.iterations, rep.neutral_directions); }
 { const Real m=1,r=0.05,k=1e6,h0=0.1; System sys; int b=sys.model.add_body(0,std::make_shared<FreeJointModel>(),Transform3(),Transform3(),RigidBodyInertia::from_solid_box(m,Vec3(0.03,0.03,0.03))); PlaneContactParams p; p.stiffness=k; p.exponent=1.5; auto c=std::make_shared<PlaneContact>(b,std::vector<ContactSphere>{{Vec3::Zero(),r}},p); sys.force_elements.push_back(c); Simulator sim(sys); sim.q(2)=r+h0;
   Real deep=0, apex=0; bool bounced=false; for(int n=0;n<40000;++n){sim.step(1e-5); Real z=sim.q(2); deep=std::max(deep,r-z); if(deep>0 && sim.states()[1].v_WB.z()>0) bounced=true; if(bounced) apex=std::max(apex,z-r);}
   printf("bounce: deepest %.6e m, energy balance rel %.3g, apex %.10f rel err %.3g\n", deep, rel(m*g_accel*(h0+deep), c->potential_energy(deep)), apex, rel(apex,h0)); }
 { const Real mp=5,mb=1,kp=2e4,L0=0.5,kc=1e6,r=0.05,t=0.02; System sys; int plate=sys.model.add_body(0,std::make_shared<PrismaticJointModel>(),Transform3(),Transform3(),RigidBodyInertia::from_solid_box(mp,Vec3(0.3,0.3,t))); int ball=sys.model.add_body(0,std::make_shared<FreeJointModel>(),Transform3(),Transform3(),RigidBodyInertia::from_solid_box(mb,Vec3(0.03,0.03,0.03)));
   sys.force_elements.push_back(std::make_shared<LinearSpringDamper>(0,plate,Vec3::Zero(),Vec3::Zero(),kp,0.0,L0)); PlaneContactParams p; p.stiffness=kc; p.friction=0.5; sys.force_elements.push_back(std::make_shared<PlaneContact>(ball,std::vector<ContactSphere>{{Vec3::Zero(),r}},p,plate,Transform3::FromTranslation(Vec3(0,0,t)))); Simulator sim(sys); sim.q(0)=L0; sim.q(3)=L0+t+r; auto rep=static_equilibrium(sim);
   printf("plate: spring err %.3g, contact depth err %.3g, iterations %d\n", std::abs(sim.q(0)-(L0-(mp+mb)*g_accel/kp)), std::abs((sim.q(0)+t+r)-sim.q(3)-mb*g_accel/kc), rep.iterations); }
}
```
