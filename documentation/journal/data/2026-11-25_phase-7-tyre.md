```matlab
function mfeval_curves(mfeval_dir, data_dir)
%MFEVAL_CURVES Writes MFeval's forces and moments for the Magic Formula test
% tyres of the multibody engine (plan task 7.4) to CSV files, which the
% engine's tests/forces/test_magic_formula_mfeval.cpp compares with its own.
%
%   mfeval_curves(mfeval_dir, data_dir)
%
% mfeval_dir: MFeval 4.3.1's MFeval folder (Marco Furlan, BSD-3, MATLAB File
% Exchange 63618), kept outside the engine's repository, read only.
% data_dir: the engine's tests/forces/data; the tyres are read from it and
% the CSV files written to its mfeval folder.
%
% MFeval is run in mode 211: no limit checks (inputs taken as they are, no
% low-speed reduction), no alpha_star (the slip angle enters as it is), no
% turn slip. A coefficient a .tir file leaves out takes the engine's default
% (0, and 2 for PKY4) instead of MFeval's sample values; each CSV lists them.
% Written for task 7.4; run with
%   matlab -batch "mfeval_curves('<mfeval_dir>', '<data_dir>')"
% from this folder.

addpath(mfeval_dir);
out_dir = fullfile(data_dir, 'mfeval');
if ~exist(out_dir, 'dir')
    mkdir(out_dir);
end
write_curves(data_dir, 'pacejka_a3_205_60R15', true, out_dir);
write_curves(data_dir, 'pacejka_a3_205_60R15_scaled', true, out_dir);
write_curves(data_dir, 'mf52_test_tyre', false, out_dir);
end

function write_curves(data_dir, name, pressure, out_dir)
tir = fullfile(data_dir, [name '.tir']);
[p, defaulted] = load_tir(tir);
X = sweep(pressure);
if pressure
    out = mfeval(p, X, 211);
else
    out = mfeval(p, X(:, 1:6), 211);
end
% Inputs: Fz kappa alpha gamma Vx p (0: the file's INFLPRES). Outputs, by
% mfeval's column: Fx 1, Fy 2, Mx 4, My 5, Mz 6, t 16, Mzr 22, mux 17,
% muy 18, Kxk 30, Kya 26, sigma_kappa 27, sigma_alpha 28.
D = [X(:, 1:4) X(:, 6) X(:, 7) out(:, [1 2 4 5 6 16 22 17 18 30 26 27 28])];
csv = fullfile(out_dir, [name '.csv']);
fid = fopen(csv, 'w');
fprintf(fid, '# MFeval 4.3.1 (Marco Furlan, BSD-3, MATLAB File Exchange 63618): mfeval(p, inputs, 211)\n');
fprintf(fid, '# for %s.tir, written by mfeval_curves.m (plan task 7.4) in MATLAB %s.\n', name, version);
fprintf(fid, '# Coefficients the file leaves out, given the engine''s defaults rather than MFeval''s:\n');
for i = 1:12:numel(defaulted)
    fprintf(fid, '#   %s\n', strjoin(defaulted(i:min(i + 11, numel(defaulted))), ' '));
end
fprintf(fid, 'Fz,kappa,alpha,gamma,Vx,p,Fx,Fy,Mx,My,Mz,t,Mzr,mux,muy,Kxk,Kya,sigma_kappa,sigma_alpha\n');
fprintf(fid, [repmat('%.12g,', 1, 18) '%.12g\n'], D');
fclose(fid);
fprintf('%s: %d rows, %d coefficients given the engine''s defaults\n', csv, size(D, 1), numel(defaulted));
end

function [p, defaulted] = load_tir(tir)
% MFeval's reading of the file, then the engine's defaults for the
% coefficients (P..., Q..., R..., S... with a number) the file does not give.
p = mfeval.readTIR(tir);
tokens = regexp(fileread(tir), '(?m)^\s*([A-Za-z_][A-Za-z0-9_]*)\s*=', 'tokens');
keys = cellfun(@(c) c{1}, tokens, 'UniformOutput', false);
defaulted = {};
names = fieldnames(p);
for i = 1:numel(names)
    n = names{i};
    if any(strcmp(keys, n)) || isempty(regexp(n, '^[PQRS][A-Z]{1,3}[0-9]+$', 'once'))
        continue;
    end
    v = 0;
    if strcmp(n, 'PKY4')
        v = 2;
    end
    if p.(n) ~= v
        p.(n) = v;
        defaulted{end + 1} = n; %#ok<AGROW>
    end
end
end

function X = sweep(pressure)
% The points: [Fz kappa alpha gamma phit Vx p].
p0 = 0;
if pressure
    p0 = 220000;
end
X = zeros(0, 7);
% Pure longitudinal slip, three loads, three inclinations (kappa above -0.94,
% where mfeval reduces the rolling resistance even in mode 211).
for Fz = [1500 4000 7000]
    for g = [-0.05 0 0.08]
        X = [X; points(Fz, linspace(-0.9, 0.9, 19)', 0, g, 16.7, p0)]; %#ok<AGROW>
    end
end
% Pure side slip.
for Fz = [1500 4000 7000]
    for g = [-0.08 0 0.05]
        X = [X; points(Fz, 0, linspace(-0.35, 0.35, 15)', g, 16.7, p0)]; %#ok<AGROW>
    end
end
% Combined slip.
[K, A] = meshgrid([-0.3 -0.1 -0.03 0.03 0.1 0.3], linspace(-0.25, 0.25, 11));
for Fz = [3000 6000]
    for g = [0 0.05]
        X = [X; points(Fz, K(:), A(:), g, 16.7, p0)]; %#ok<AGROW>
    end
end
% Forward speed: the rolling resistance's speed terms.
X = [X; points(4000, 0.02, 0.03, 0.02, [2 5 10 20 30 40]', p0)];
% Inflation pressure (MF 6.1).
if pressure
    for pi = [180000 260000]
        X = [X; points(4000, linspace(-0.4, 0.4, 9)', 0, 0.03, 16.7, pi)]; %#ok<AGROW>
        X = [X; points(4000, 0, linspace(-0.2, 0.2, 9)', 0.03, 16.7, pi)]; %#ok<AGROW>
        X = [X; points(4000, [-0.1 0.05 0.1]', [0.1 -0.05 0.1]', 0.03, 16.7, pi)]; %#ok<AGROW>
    end
end
end

function R = points(Fz, kappa, alpha, gamma, Vx, p)
n = max([numel(kappa) numel(alpha) numel(Vx)]);
c = @(x) x(:) .* ones(n, 1);
R = [c(Fz) c(kappa) c(alpha) c(gamma) zeros(n, 1) c(Vx) c(p)];
end
```
