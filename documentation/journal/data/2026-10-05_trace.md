## Programs

```cpp
#include <cstdio>
#include <cmath>
#include "mbd/kernel/diagnostics.hpp"
#include "mbd/kernel/joint_forces.hpp"
using namespace mbd; using namespace mbd::kernel;
int main(){
  System sys; sys.model.gravity=Vec3(0,-g_accel,0); auto I=RigidBodyInertia::from_solid_box(1.0,Vec3(0.02,0.02,0.02)); I.com_B=Vec3(0,-0.5,0);
  int b=sys.model.add_body(0,std::make_shared<RevoluteJointModel>(),Transform3(),Transform3(),I,"bob"); JointCoordinateForceParams p; p.damper=Curve::linear(0.3); sys.joint_forces.push_back(std::make_shared<JointCoordinateForce>(sys.model,b,p));
  Simulator sim(sys); sim.q(0)=0.5; sim.trace=true; sim.run(5.0,1e-3);
  double worst=0; for(auto& r: sim.trace_rows()) worst=std::max(worst,std::abs(r.balance));
  const auto& last=sim.trace_rows().back(); const auto& first=sim.trace_rows().front();
  printf("damped pendulum 5 s at 1 ms: energy %.6f -> %.6f J, damper work %.6f J, worst |balance| %.3g J\n", first.kinetic+first.potential, last.kinetic+last.potential, last.work, worst);
  printf("diagnosis: %s", diagnose(sim.trace_rows()).summary().c_str());
}
```
