Reading it: the tyre loads sum to the model's weight to 1e-7 N and their
moments about the centre of mass are below 2e-7 N m. `set_vehicle_equilibrium`
puts the chassis 2.1 cm lower than the equilibrium. The second car, started
from the same state and simulated with RK4 at 1 ms, approaches the static
state (chassis height within 3e-5 m, suspension angles within 2e-4 rad after
8 s) but keeps rolling backwards at about 4 mm/s: its free-rolling tyres have
no rolling resistance (finding F7), so the momentum of the initial transient
never dies away. The time includes the final stiffness evaluation for the
report.

## Program

```cpp
#include <cstdio>
#include <chrono>
#include "mbd/kernel/statics.hpp"
#include "mbd/kernel/algorithms.hpp"
#include "mbd/vehicle/vehicle_template.hpp"
using namespace mbd; using namespace mbd::kernel;
int main(){
  auto tmpl = VehicleTemplate::DefaultSedan();
  tmpl.front_axle.suspension_type = SuspensionType::DoubleWishbone;
  tmpl.rear_axle.suspension_type  = SuspensionType::DoubleWishbone;
  System sys; auto vh = build_vehicle(sys, tmpl);
  Simulator sim(sys); set_vehicle_equilibrium(sim, vh); sim.initialize();
  const VecX q_start = sim.q;
  auto t0=std::chrono::steady_clock::now();
  auto r = static_equilibrium(sim);
  auto t1=std::chrono::steady_clock::now();
  printf("%s", r.summary(sys.model).c_str());
  printf("statics time %.1f ms\n", std::chrono::duration<double,std::milli>(t1-t0).count());
  const Model& model=sys.model;
  sim.acceleration(sim.q, sim.v, sim.time);
  double mass=0; for(int i=1;i<model.nbodies();++i) mass+=model.inertia[i].mass;
  Vec3 com=center_of_mass(model, sim.data());
  double load=0,pitch=0,roll=0;
  for(int c=0;c<4;++c){auto* t=vh.tire(c); load+=t->last_Fz; pitch+=t->last_Fz*(t->last_contact_pos_W.x()-com.x()); roll+=t->last_Fz*(t->last_contact_pos_W.z()-com.z()); printf("tyre %d Fz %.3f N at x %.4f z %.4f\n",c,t->last_Fz,t->last_contact_pos_W.x(),t->last_contact_pos_W.z());}
  printf("mass %.3f kg weight %.4f N loads %.4f N diff %.3g; pitch %.3g N m roll %.3g N m; com (%.5f %.5f %.5f)\n",mass,mass*g_accel,load,load-mass*g_accel,pitch,roll,com.x(),com.y(),com.z());
  const int ic = model.idx_q[vh.chassis_body];
  VecX q_static = sim.q;
  printf("static chassis y %.9f, start y %.9f\n", q_static(ic+1), q_start(ic+1));
  // Dynamic settle from the same start
  System sys2; auto vh2 = build_vehicle(sys2, tmpl);
  Simulator s2(sys2); set_vehicle_equilibrium(s2, vh2); s2.initialize();
  for (double T : {1.0, 2.0, 4.0, 8.0}) {
     while (s2.time < T - 1e-9) s2.step(1e-3);
     VecX a = s2.acceleration(s2.q, VecX::Zero(model.nv), s2.time);
     // compare suspension and heave/pitch/roll: all coordinates except chassis x,z translation and yaw
     VecX d(model.nv); difference(model, q_static, s2.q, d);
     printf("t=%.0f s: chassis y %.9f (static %.9f, diff %.3g); |v| %.3g; chassis x drift %.3g; max |dq| non-chassis %.3g\n", T, s2.q(ic+1), q_static(ic+1), s2.q(ic+1)-q_static(ic+1), s2.v.cwiseAbs().maxCoeff(), s2.q(ic)-q_static(ic), d.tail(model.nv-6).cwiseAbs().maxCoeff());
  }
}
```
